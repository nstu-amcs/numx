\section{Постановка задачи}

\noindent Построить МКЭ для уравнения параболического типа

\begin{align}
  \label{eq:1} -\nabla(\lambda \nabla u) + \gamma u + 
    \sigma\frac{\partial u}{\partial t} = f
\end{align}

\vspace{4mm}
\noindent в декартовой системе координат

\begin{align}
  - \frac{\partial}{\partial x}(\lambda\frac{\partial u}{\partial x}) 
  - \frac{\partial}{\partial y}(\lambda\frac{\partial u}{\partial y}) 
  - \frac{\partial}{\partial z}(\lambda\frac{\partial u}{\partial z})
    + \gamma u + \sigma\frac{\partial u}{\partial t} = f
\end{align}

\vskip 4mm
\noindent с учётом следующих краевых условий:

\begin{align}
  &u \big|_{S_1} = u_g \\
  &\lambda\frac{\partial u}{\partial n} \bigg|_{S_2} = \theta \\
  &\lambda\frac{\partial u}{\partial n} \bigg|_{S_3} +
    \beta(u \big|_{S_3} - u_{\beta}) = 0
\end{align}

\section{Теоретическая часть}

\subsection{Вариационная постановка}

\noindent Эквивалентная вариационная постановка в форме уравнения Галёркина для
эллиптического уравнения, входящего в уравнение \ref{eq:1}:

\begin{align} \label{eq:2}
  \begin{split}
    \int_{\Omega}\lambda \nabla u \nabla v_0 d\Omega &+
    \int_{\Omega}\gamma uv_0 d\Omega +
    \int_{S_3}\beta uv_0 dS \\
    &= \int_{\Omega}f v_0 d\Omega +
    \int_{S_2}\theta v_0 dS +
    \int_{S_3}\beta u_{\beta} v_0 dS, \ \ \forall v_0 \in H_0^1
  \end{split}
\end{align}

\vskip 5mm
\noindent Аппроксимация уравнения \ref{eq:2} на конечномерных подпространствах $V^h_g$ и $V^h_0$ 
получается заменой функций $u \in H^1_g$ и $v_0 \in H^1_0$ на 
функции $u_h \in V^h_g$ и $v^h_0 \in V^h_0$ соответственно:

\begin{align}
  \begin{split}
    \int_{\Omega}\lambda \nabla u^h \nabla v^h_0 d\Omega &+
    \int_{\Omega}\gamma u^h v^h_0 d\Omega +
    \int_{S_3}\beta u^h v^h_0 dS \\
    &= \int_{\Omega}f v^h_0 d\Omega +
    \int_{S_2}\theta v^h_0 dS +
    \int_{S_3}\beta u_{\beta} v^h_0 dS, \ \ \forall v^h_0 \in V^h_0
  \end{split}
\end{align}

\vskip 5mm

\subsection{Конечноэлементная СЛАУ}

\noindent Раскладывая функции $u$ и $v_0$ по базису, переходим к конечноэлементной СЛАУ

\begin{align}
  \begin{split}
    \sum_{j=1}^{n}\big(
      \int_{\Omega}\lambda \nabla \psi_j \nabla \psi_i d\Omega &+
      \int_{\Omega}\gamma \psi_j \psi_i d\Omega +
      \int_{S_3}\beta \psi_j \psi_i dS
    \big)q_j \\
    &= \int_{\Omega}f \psi_i d\Omega +
    \int_{S_2}\theta \psi_i dS +
    \int_{S_3}\beta u_{\beta} \psi_i dS, \ \ i \in N_0
  \end{split}
\end{align}

\vskip 5mm
\noindent или в матричном виде

\begin{align}
  (\textbf{G} + \textbf{M}^{\gamma} + \textbf{M}^{S_3})\textbf{q} = \textbf{b}
\end{align}

\vskip 5mm
\noindent где компоненты определяются соотношениями

\begin{align}
  &G_{ij} = \int_{\Omega}\lambda \nabla \psi_j \nabla \psi_i d\Omega
          = \sum_{k}\int_{\hat{\Omega}}\hat{\lambda}\nabla\hat{\psi}_j\nabla\hat{\psi}_i d\hat{\Omega} \\
  &M^{\gamma}_{ij} = \int_{\Omega}\gamma \psi_j \psi_i d\Omega
                   = \sum_{k}\int_{\hat{\Omega}}\hat{\gamma}\hat{\psi}_j\hat{\psi}_i d\hat{\Omega}\\
  &M^{S_3}_{ij} = \int_{S_3}\beta \psi_j \psi_i dS
                = \sum_{k}\int_{\hat{S}_3}\hat{\beta}\hat{\psi}_j\hat{\psi}_i d\hat{S} \\
  &b^{\Omega}_{i} = \int_{\Omega}f \psi_i d\Omega
                  = \sum_{k}\int_{\hat{\Omega}}\hat{f}\hat{\psi}_i d\hat{\Omega} \\
  &b^{S_2}_{i} = \int_{S_2}\theta \psi_i dS
               = \sum_{k}\int_{\hat{S}_2}\hat{\theta}\hat{\psi}_i d\hat{S} \\
  &b^{S_3}_{i} = \int_{S_3}\beta u_{\beta} \psi_i dS
               = \sum_{k}\int_{\hat{S}_3}\hat{\beta}\hat{u}_{\beta}\hat{\psi}_i d\hat{S}
\end{align}

\vskip 5mm

\subsection{Трилинейные базисные функции}

\vskip 5mm
\noindent Трилинейные бизисные функции на прямоугольном параллелепипеде $\Omega_{psr}$ строятся
  следующим образом:

\begin{align*}
  & X_1(x) = \frac{x_{p+1}-x}{h_x}, & &X_2(x) = \frac{x - x_{p}}{h_x}, & &h_x = x_{p+1} - x_p, \\
  & Y_1(y) = \frac{y_{s+1}-y}{h_y}, & &Y_2(y) = \frac{y - y_{s}}{h_y}, & &h_y = y_{s+1} - y_s, \\
  & Z_1(z) = \frac{z_{r+1}-z}{h_z}, & &Z_2(z) = \frac{z - z_{r}}{h_z}, & &h_z = z_{r+1} - z_r.
\end{align*}

\begin{align}
  \label{eq:3} \hat{\psi}_i = X_{\mu(i)}Y_{\nu(i)}Z_{\vartheta(i)}
\end{align}

\vskip 5mm
\noindent Выражения для вычисления компонент локальных матриц и векторов на прямоугольном
параллелепипеде $\Omega_{psr}$:

\begin{align}
  \begin{split}
    \hat{G}_{ij} = \hat{\lambda}(
      &G^{x}_{\mu(k)\mu(j)\mu(i)}M^{y}_{\nu(k)\nu(j)\nu(i)}M^{z}_{\vartheta(k)\vartheta(j)\vartheta(i)} + \\
      &M^{x}_{\mu(k)\mu(j)\mu(i)}G^{y}_{\nu(k)\nu(j)\nu(i)}M^{z}_{\vartheta(k)\vartheta(j)\vartheta(i)} + \\
      &M^{x}_{\mu(k)\mu(j)\mu(i)}M^{y}_{\nu(k)\nu(j)\nu(i)}G^{z}_{\vartheta(k)\vartheta(j)\vartheta(i)}
    )
  \end{split}
\end{align}

\begin{align}
  &\hat{M}^{\gamma}_{ij} = \hat{\gamma}(
      M^{x}_{\mu(j)\mu(i)}M^{y}_{\nu(j)\nu(i)}M^{z}_{\vartheta(j)\vartheta(i)}
  ) \\
  &\hat{M}^{S_3}_{ij} = \hat{\beta}(
      M^{\xi}_{\mu(j)\mu(i)}M^{\chi}_{\nu(j)\nu(i)}
  ) \\
  &\hat{b}^{\Omega}_{i} = \sum_{k=1}^{8}\hat{f}_{k}(
      M^{x}_{\mu(k)\mu(i)}M^{y}_{\nu(k)\nu(i)}M^{z}_{\vartheta(k)\vartheta(i)}
  ) \\
  &\hat{b}^{S_2}_{i} = \sum_{k=1}^{4}\hat{\theta}_{k}(
      M^{\xi}_{\mu(k)\mu(i)}M^{\chi}_{\nu(k)\nu(i)}
  ) \\
  &\hat{b}^{S_3}_{i} = \hat{\beta}\sum_{k=1}^{4}\hat{u}_{\beta, k}(
      M^{\xi}_{\mu(k)\mu(i)}M^{\chi}_{\nu(k)\nu(i)}
  )
\end{align}

\vskip 5mm
\noindent где

\begin{align}
  &\textbf{G}^{\xi} = \frac{1}{h_{\xi}}
    \begin{pmatrix}
      1 & -1 \\
      -1 & 1
    \end{pmatrix} \\
  &\textbf{M}^{\xi} = \frac{h_{\xi}}{6}
    \begin{pmatrix}
      2 & 1 \\
      1 & 2
    \end{pmatrix}
\end{align}

\subsection{Аппроксимация по времени}

\subsubsection{Двухслойная схема}

\noindent Полагая, что коэффициенты $\lambda$ и $\gamma$ не зависят от времени, неявная 
  двухслойная схема аппроксимации уравнения \ref{eq:1} по времени может быть выписана как

\begin{align}
  -\nabla(\lambda \nabla u^j) + \gamma u^j + 
    \sigma\frac{u^j - u^{j-1}}{\Delta t} = f^j, \ \ j = 1...J
\end{align}

\vspace{5mm}
\noindent где $u^{j-1}$ - решение, полученное на предыдущем временном слое, а $u^0$ - начальное условие.

\vspace{2mm}
\noindent Проводя преобразования, аналогичные тем, что были проведены для эллиптической составляющей, 
  получаем следующее матричное уравнение:

\begin{align}
  (\textbf{G} + \textbf{M}^{\gamma} + \textbf{M}^{S_3} + \frac{1}{\Delta t}\textbf{M}^{\sigma})\textbf{q}^j =
    \textbf{b}^j + \frac{1}{\Delta t}\textbf{M}^{\sigma}\textbf{q}^{j-1}
\end{align}

\vspace{3mm}

\subsubsection{Трехслойная схема}

\noindent Представим искомое решение $u$ на интервале $(t_{j-2}, t_j)$ в следующем виде:

\begin{align}
  \label{eq:4} u(x, y, z, t) \approx u^{j-2}(x, y, z)\eta^j_2(t) + 
                        u^{j-1}(x, y, z)\eta^j_1(t) +
                        u^{j}(x, y, z)\eta^j_0(t)
\end{align}

\vspace{5mm}
\noindent где функции $\eta^j_2(t), \eta^j_1(t), \eta^j_0(t)$ - базисные квадратичные полиномы
  Лагранжа, которые могут быть записаны в виде

\begin{align}
  &\eta^j_2(t) = \frac{(t - t_{j-1})(t - t_j)}{2\Delta t^2} \\
  &\eta^j_1(t) = -\frac{(t - t_{j-2})(t - t_j)}{\Delta t^2} \\
  &\eta^j_0(t) = \frac{(t - t_{j-2})(t - t_{j-1})}{2\Delta t^2}
\end{align}

\vspace{5mm}
\noindent Производные по времени на $j$-ом временном слое:

\begin{align}
  &\frac{d\eta^j_2(t)}{dt}\bigg|_{t = t_j} = \frac{1}{2 \Delta t} \\
  &\frac{d\eta^j_1(t)}{dt}\bigg|_{t = t_j} = -\frac{2}{\Delta t} \\
  &\frac{d\eta^j_0(t)}{dt}\bigg|_{t = t_j} = \frac{3}{2 \Delta t}
\end{align}

\vspace{5mm}
\noindent Применяя выражение \ref{eq:4} для аппроксимации производной по времени уравнения \ref{eq:1}
  на $j$-ом временном слое:

\begin{align}
  \begin{split}
    \frac{\partial}{\partial t}(
      &u^{j-2}(x, y, z)\eta^j_2(t) + 
      u^{j-1}(x, y, z)\eta^j_1(t) +
      u^{j}(x, y, z)\eta^j_0(t)
    )\bigg|_{t = t_j} \\ 
    &- \nabla(\lambda \nabla u^j) + \gamma u^j = f^j, \ \ j = 2...J
  \end{split}
\end{align}

\vspace{5mm}
\noindent или, с учётом выражений для вычисления производных

\begin{align}
  -\nabla(\lambda \nabla u^j) + \gamma u^j + 
    \frac{1}{2 \Delta t} u^{j-2} -
    \frac{2}{\Delta t} u^{j-1} + 
    \frac{3}{2 \Delta t} u^j = f^j
\end{align}

\vspace{5mm}
\noindent Проводя преобразования, аналогичные тем, что были проведены для эллиптической составляющей, 
  получаем следующее матричное уравнение:

\begin{align}
  (\textbf{G} + \textbf{M}^{\gamma} + \textbf{M}^{S_3} + 
    \frac{3}{2 \Delta t}\textbf{M}^{\sigma})\textbf{q}^j = \textbf{b}^j 
      - \frac{1}{2 \Delta t}\textbf{M}^{\sigma}\textbf{q}^{j-2}
      + \frac{2}{\Delta t}\textbf{M}^{\sigma}\textbf{q}^{j-1}
\end{align}

\vspace{3mm}

\subsubsection{Четырехслойная схема}

\noindent Представим искомое решение $u$ на интервале $(t_{j-2}, t_j)$ в следующем виде:

\begin{align}
  \label{eq:5} u(x, y, z, t) \approx u^{j-3}(x, y, z)\eta^j_3(t) +
                        u^{j-2}(x, y, z)\eta^j_2(t) + 
                        u^{j-1}(x, y, z)\eta^j_1(t) +
                        u^{j}(x, y, z)\eta^j_0(t)
\end{align}

\vspace{5mm}
\noindent где функции $\eta^j_3(t), \eta^j_2(t), \eta^j_1(t), \eta^j_0(t)$ - базисные кубические полиномы
  Лагранжа, которые могут быть записаны в виде

\begin{align}
  &\eta^j_3(t) = -\frac{(t - t_{j-2})(t - t_{j-1})(t - t_{j})}{6\Delta t^3} \\
  &\eta^j_2(t) = \frac{(t - t_{j-3})(t - t_{j-1})(t - t_{j})}{2\Delta t^3} \\
  &\eta^j_1(t) = -\frac{(t - t_{j-3})(t - t_{j-2})(t - t_{j})}{2\Delta t^3} \\
  &\eta^j_0(t) = \frac{(t - t_{j-3})(t - t_{j-2})(t - t_{j-1})}{6\Delta t^3}
\end{align}

\vspace{5mm}
\noindent Производные по времени на $j$-ом временном слое:

\begin{align}
  &\frac{d\eta^j_3(t)}{dt}\bigg|_{t = t_j} = -\frac{1}{3 \Delta t} \\
  &\frac{d\eta^j_2(t)}{dt}\bigg|_{t = t_j} = \frac{3}{2 \Delta t} \\
  &\frac{d\eta^j_1(t)}{dt}\bigg|_{t = t_j} = -\frac{3}{\Delta t} \\
  &\frac{d\eta^j_0(t)}{dt}\bigg|_{t = t_j} = \frac{11}{6 \Delta t}
\end{align}

\vspace{5mm}
\noindent Применяя выражение \ref{eq:5} для аппроксимации производной по времени уравнения \ref{eq:1}
  на $j$-ом временном слое:

\begin{align}
  \begin{split}
    \frac{\partial}{\partial t}(
      &u^{j-3}(x, y, z)\eta^j_3(t) + 
      u^{j-2}(x, y, z)\eta^j_2(t) + 
      u^{j-1}(x, y, z)\eta^j_1(t) +
      u^{j}(x, y, z)\eta^j_0(t)
    )\bigg|_{t = t_j} \\ 
    &- \nabla(\lambda \nabla u^j) + \gamma u^j = f^j, \ \ j = 2...J
  \end{split}
\end{align}

\vspace{5mm}
\noindent или, с учётом выражений для вычисления производных

\begin{align}
  -\nabla(\lambda \nabla u^j) + \gamma u^j - 
    \frac{1}{3 \Delta t} u^{j-3} +
    \frac{3}{2 \Delta t} u^{j-2} -
    \frac{3}{\Delta t} u^{j-1} + 
    \frac{11}{6 \Delta t} u^j = f^j
\end{align}

\vspace{5mm}
\noindent Проводя преобразования, аналогичные тем, что были проведены для эллиптической составляющей, 
  получаем следующее матричное уравнение:

\begin{align}
  (\textbf{G} + \textbf{M}^{\gamma} + \textbf{M}^{S_3} + 
    \frac{11}{6 \Delta t}\textbf{M}^{\sigma})\textbf{q}^j = \textbf{b}^j 
      + \frac{1}{3 \Delta t} \textbf{M}^{\sigma}\textbf{q}^{j-3}
      - \frac{3}{2 \Delta t} \textbf{M}^{\sigma}\textbf{q}^{j-2}
      + \frac{3}{\Delta t} \textbf{M}^{\sigma}\textbf{q}^{j-1}
\end{align}
