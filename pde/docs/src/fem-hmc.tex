\section{Постановка задачи}

\noindent Построить МКЭ для нелинейного уравнения гиперболического типа

\begin{align}
  \label{eq:1} -\nabla(\lambda\nabla u) 
    + \sigma\frac{\partial u}{\partial t}
    + \chi\frac{\partial^2u}{\partial t^2} = f,
\end{align}

\vskip 4mm
\noindent в котором правая часть $f$ представима в виде

\begin{align}
  f(x, y, z, t) = f^s(x, y, z)sin(\omega t) + f^c(x, y, z)cos(\omega t),
\end{align}

\vskip 4mm
\noindent с учётом следующих краевых условий:

\begin{align}
  \lambda\frac{\partial u^s}{\partial n} \bigg|_{S_2} = \theta^s,\ \  &
  \lambda\frac{\partial u^c}{\partial n} \bigg|_{S_2} = \theta^c \\
  \lambda\frac{\partial u^s}{\partial n} \bigg|_{S_3} + \beta(u^s|_{S_3} - u^s_{\beta}) = 0,\ \  &
  \lambda\frac{\partial u^c}{\partial n} \bigg|_{S_3} + \beta(u^c|_{S_3} - u^c_{\beta}) = 0.
\end{align}

\section{Теоретическая часть}

\noindent Решение уравнения \ref{eq:1} может быть представлено в следующем виде:

\begin{align}
  \label{eq:2} u(x, y, z, t) = u^s(x, y, z)sin(\omega t) + u^c(x, y, z)cos(\omega t).
\end{align}

\vskip 4mm
\noindent Подставляя \ref{eq:2} в \ref{eq:1} и группируя по $sin(\omega t)$ и $cos(\omega t)$,
  получаем систему уравнений

\begin{align}\label{eq:3}
  \begin{cases}
    -\nabla(\lambda\nabla u^s) - \omega\sigma u^c - \omega^2\chi u^s = f^s, \\
    -\nabla(\lambda\nabla u^c) + \omega\sigma u^s - \omega^2\chi u^c = f^c.
  \end{cases}
\end{align}

\vskip 4mm
\noindent Эквивалентная вариационная постановка в форме уравнения Галёркина для
  системы уравнений \ref{eq:3}:

\begin{align}
  \begin{cases}
    \begin{split}
      \int_{\Omega}(\lambda\nabla u^s \nabla v_0 & - \omega\sigma u^c v_0 - \omega^2\chi u^s v_0) d\Omega +
      \int_{S_3}\beta u^s v_0 dS \\
      & = \int_{\Omega}f^s v_0 d\Omega +
          \int_{S_2}\theta^s v_0 dS +
          \int_{S_3}\beta u^s_{\beta} v_0 dS,
    \end{split} \\
    \begin{split}
      \int_{\Omega}(\lambda\nabla u^c \nabla v_0 & - \omega\sigma u^s v_0 - \omega^2\chi u^c v_0) d\Omega +
      \int_{S_3}\beta u^c v_0 dS \\
      & = \int_{\Omega}f^c v_0 d\Omega +
          \int_{S_2}\theta^c v_0 dS +
          \int_{S_3}\beta u^c_{\beta} v_0 dS,
    \end{split}
  \end{cases}
\end{align}

\vskip 5mm
\noindent Раскладывая функции $u^s$, $u^c$ и $v_0$ по базису, переходим 
к конечноэлементной системе из $2n$ уравнений c $2n$ неизвестными:

\begin{align}
  \begin{cases}
    \begin{split}
      \sum_{j=1}^{n}\bigg(
        \int_{\Omega}(\lambda\nabla \psi_j \nabla \psi_i & - \omega^2\chi\psi_j\psi_i)d\Omega +
        \int_{S_3}\beta\psi_j\psi_i dS\bigg
      )q^s_j - \omega\sum_{j=1}^{n}\bigg(\int_{\Omega}\sigma\psi_j\psi_i d\Omega\bigg)q^c_j \\
      & = \int_{\Omega}f^s \psi_i d\Omega +
          \int_{S_2}\theta^s \psi_i dS +
          \int_{S_3}\beta u^s_{\beta} \psi_i dS,\ \ i=1..n
    \end{split} \\
    \begin{split}
      \sum_{j=1}^{n}\bigg(
        \int_{\Omega}(\lambda\nabla \psi_j \nabla \psi_i & - \omega^2\chi\psi_j\psi_i)d\Omega +
        \int_{S_3}\beta\psi_j\psi_i dS\bigg
      )q^c_j - \omega\sum_{j=1}^{n}\bigg(\int_{\Omega}\sigma\psi_j\psi_i d\Omega\bigg)q^s_j \\
      & = \int_{\Omega}f^c \psi_i d\Omega +
          \int_{S_2}\theta^c \psi_i dS +
          \int_{S_3}\beta u^c_{\beta} \psi_i dS\ \ i=1..n
    \end{split}
  \end{cases}
\end{align}

\vskip 5mm
\noindent Вид трилинейных базисных функций:

\begin{align*}
  & X_1(x) = \frac{x_{p+1}-x}{h_x}, & &X_2(x) = \frac{x - x_{p}}{h_x}, & &h_x = x_{p+1} - x_p, \\
  & Y_1(y) = \frac{y_{s+1}-y}{h_y}, & &Y_2(y) = \frac{y - y_{s}}{h_y}, & &h_y = y_{s+1} - y_s, \\
  & Z_1(z) = \frac{z_{r+1}-z}{h_z}, & &Z_2(z) = \frac{z - z_{r}}{h_z}, & &h_z = z_{r+1} - z_r.
\end{align*}

\begin{align}
  \psi_i = X_{\mu(i)}Y_{\nu(i)}Z_{\vartheta(i)}.
\end{align}
